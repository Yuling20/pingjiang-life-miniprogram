Page({
  data: {
    // 按 Claude 要求追加：
    profileBg: '',           // VIP自定义背景图
    isSelf: false,           // 是否是自己的主页（mine跳来时为true）
    // 原有字段：
    userInfo: {
      nickName:  '平江用户',
      avatarUrl: '/images/avatars男/avatar01.png',
      bio:       '',
      isVip:     false,
    },
    defaultBg: '',           // 默认背景（纯色渐变，不需要图片）
    stats: {
      follow: 20,
      fans:   15,
      posts:  5,
      likes:  48,
    },
    postList: [],
    isLoading: true,
  },

  onLoad(options) {
    // 按 Claude 要求追加：
    const isSelf = options.self === '1';
    this.setData({ isSelf });

    // 同步"我的"页面的 userInfo（头像、昵称、简介）
    try {
      const info = wx.getStorageSync('userInfo');
      if (info && info.nickName) {
        this.setData({
          // 只覆盖这几个展示字段，不影响原有userId等逻辑
          nickName:  info.nickName,
          avatarUrl: info.avatarUrl,
          bio:       info.bio || '',
          isVip:     !!info.isVip,
        });
      }
    } catch(e) {}

    // 读取VIP背景图
    try {
      const bg = wx.getStorageSync('profileBgImage') || '';
      this.setData({ profileBg: bg });
    } catch(e) {}

    // 原有逻辑
    this._loadUserInfo();
    this._loadProfileBg();
    this._loadStats();
    this._loadPosts();
  },

  onShow() {
    // 每次显示时重新同步，保证简介实时更新
    this._loadUserInfo();
    this._loadProfileBg();
  },

  // ✅ 问题2：从 userInfo 同步简介（与"我的"页面完全一致）
  _loadUserInfo() {
    try {
      const info = wx.getStorageSync('userInfo');
      if (info && info.nickName) {
        this.setData({ userInfo: info });
      }
    } catch(e) {}
  },

  // ✅ VIP背景：从 storage 读取用户自定义背景
  _loadProfileBg() {
    try {
      const bg = wx.getStorageSync('profileBgImage') || '';
      this.setData({ profileBg: bg });
    } catch(e) {}
  },

  _loadStats() {
    try {
      const s = wx.getStorageSync('profileStats') || {};
      this.setData({
        stats: {
          follow: s.follow || 20,
          fans:   s.fans   || 15,
          posts:  (wx.getStorageSync('myPublishedPosts') || []).length || 5,
          likes:  s.likes  || 48,
        }
      });
    } catch(e) {}
  },

  _loadPosts() {
    try {
      const list = wx.getStorageSync('myPublishedPosts') || [];
      // 格式化时间显示
      const formatted = list.map(p => ({
        ...p,
        timeLabel: this._formatTime(p.createTime || p.time || Date.now()),
      }));
      this.setData({ postList: formatted, isLoading: false });
    } catch(e) {
      this.setData({ isLoading: false });
    }
  },

  _formatTime(ts) {
    const now  = Date.now();
    const diff = now - Number(ts);
    if (diff < 60 * 1000)           return '刚刚';
    if (diff < 60 * 60 * 1000)      return `${Math.floor(diff/60000)}分钟前`;
    if (diff < 24 * 60 * 60 * 1000) return `${Math.floor(diff/3600000)}小时前`;
    const d = new Date(Number(ts));
    return `${d.getMonth()+1}月${d.getDate()}日`;
  },

  // ─── VIP换背景 ─────────────────────────────────────────────
  onChangeBg() {
    const info = this.data.userInfo;
    if (!info.isVip) {
      wx.showModal({
        title: 'VIP专属权益',
        content: '自定义贴吧主页背景图是VIP专属功能，开通VIP后即可使用。',
        confirmText: '去开通',
        cancelText: '取消',
        success: (res) => {
          if (res.confirm) {
            wx.navigateBack();
            // 返回"我的"页面后可点击"开通VIP"
          }
        }
      });
      return;
    }
    wx.chooseMedia({
      count: 1,
      mediaType: ['image'],
      sizeType: ['compressed'],
      sourceType: ['album', 'camera'],
      success: (res) => {
        const path = res.tempFiles[0].tempFilePath;
        wx.setStorageSync('profileBgImage', path);
        this.setData({ profileBg: path });
        wx.showToast({ title: '背景已更新', icon: 'success' });
      }
    });
  },

  // ─── 帖子互动（点赞/评论跳转） ────────────────────────────
  onLikePost(e) {
    const idx = e.currentTarget.dataset.idx;
    const list = this.data.postList;
    list[idx].liked   = !list[idx].liked;
    list[idx].likeCount = (list[idx].likeCount || 0) + (list[idx].liked ? 1 : -1);
    this.setData({ postList: list });
  },

  goPostDetail(e) {
    const id = e.currentTarget.dataset.id;
    if (!id) return;
    wx.navigateTo({
      url: `/pages/community/detail/index?id=${id}`,
      fail: () => wx.showToast({ title: '跳转失败', icon: 'none' })
    });
  },

  goBack() {
    wx.navigateBack();
  },
});{

  "usingComponents": {}
}

    const isSelf = options.self === '1';
    this.setData({ isSelf });

    // 同步"我的"页面的 userInfo（头像、昵称、简介）
    try {
      const info = wx.getStorageSync('userInfo');
      if (info && info.nickName) {
        this.setData({
          // 只覆盖这几个展示字段，不影响原有userId等逻辑
          nickName:  info.nickName,
          avatarUrl: info.avatarUrl,
          bio:       info.bio || '',
          isVip:     !!info.isVip,
        });
      }
    } catch(e) {}

    // 读取VIP背景图
    try {
      const bg = wx.getStorageSync('profileBgImage') || '';
      this.setData({ profileBg: bg });
    } catch(e) {}

    // 原有逻辑
    this._loadUserInfo();
    this._loadProfileBg();
    this._loadStats();
    this._loadPosts();
  },

  onShow() {
    // 每次显示时重新同步，保证简介实时更新
    this._loadUserInfo();
    this._loadProfileBg();
  },

  // ✅ 问题2：从 userInfo 同步简介（与"我的"页面完全一致）
  _loadUserInfo() {
    try {
      const info = wx.getStorageSync('userInfo');
      if (info && info.nickName) {
        this.setData({ userInfo: info });
      }
    } catch(e) {}
  },

  // ✅ VIP背景：从 storage 读取用户自定义背景
  _loadProfileBg() {
    try {
      const bg = wx.getStorageSync('profileBgImage') || '';
      this.setData({ profileBg: bg });
    } catch(e) {}
  },

  _loadStats() {
    try {
      const s = wx.getStorageSync('profileStats') || {};
      this.setData({
        stats: {
          follow: s.follow || 20,
          fans:   s.fans   || 15,
          posts:  (wx.getStorageSync('myPublishedPosts') || []).length || 5,
          likes:  s.likes  || 48,
        }
      });
    } catch(e) {}
  },

  _loadPosts() {
    try {
      const list = wx.getStorageSync('myPublishedPosts') || [];
      // 格式化时间显示
      const formatted = list.map(p => ({
        ...p,
        timeLabel: this._formatTime(p.createTime || p.time || Date.now()),
      }));
      this.setData({ postList: formatted, isLoading: false });
    } catch(e) {
      this.setData({ isLoading: false });
    }
  },

  _formatTime(ts) {
    const now  = Date.now();
    const diff = now - Number(ts);
    if (diff < 60 * 1000)           return '刚刚';
    if (diff < 60 * 60 * 1000)      return `${Math.floor(diff/60000)}分钟前`;
    if (diff < 24 * 60 * 60 * 1000) return `${Math.floor(diff/3600000)}小时前`;
    const d = new Date(Number(ts));
    return `${d.getMonth()+1}月${d.getDate()}日`;
  },

  // ─── VIP换背景 ─────────────────────────────────────────────
  onChangeBg() {
    const info = this.data.userInfo;
    if (!info.isVip) {
      wx.showModal({
        title: 'VIP专属权益',
        content: '自定义贴吧主页背景图是VIP专属功能，开通VIP后即可使用。',
        confirmText: '去开通',
        cancelText: '取消',
        success: (res) => {
          if (res.confirm) {
            wx.navigateBack();
            // 返回"我的"页面后可点击"开通VIP"
          }
        }
      });
      return;
    }
    wx.chooseMedia({
      count: 1,
      mediaType: ['image'],
      sizeType: ['compressed'],
      sourceType: ['album', 'camera'],
      success: (res) => {
        const path = res.tempFiles[0].tempFilePath;
        wx.setStorageSync('profileBgImage', path);
        this.setData({ profileBg: path });
        wx.showToast({ title: '背景已更新', icon: 'success' });
      }
    });
  },

  // ─── 帖子互动（点赞/评论跳转） ────────────────────────────
  onLikePost(e) {
    const idx = e.currentTarget.dataset.idx;
    const list = this.data.postList;
    list[idx].liked   = !list[idx].liked;
    list[idx].likeCount = (list[idx].likeCount || 0) + (list[idx].liked ? 1 : -1);
    this.setData({ postList: list });
  },

  goPostDetail(e) {
    const id = e.currentTarget.dataset.id;
    if (!id) return;
    wx.navigateTo({
      url: `/pages/community/detail/index?id=${id}`,
      fail: () => wx.showToast({ title: '跳转失败', icon: 'none' })
    });
  },

  goBack() {
    wx.navigateBack();
  },
